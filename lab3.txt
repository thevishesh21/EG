// Numerical and Analytical Solution of ODE

clc;
clf();

function ydot = f(t,y)
    ydot = 5*t - t*y;
endfunction

x0 = 2;
y0 = 1;

t = 2:0.01:5;

ydiff = ode(y0,x0,t,f);

// Numerical Solution
subplot(1,2,1)
plot2d(t,ydiff,style=color('red'))
xlabel('x')
ylabel('y')
title('Numerical solution of ODE')

// Analytical Solution
ysol = 5 - 4*exp(2-(t.^2/2));

subplot(1,2,2)
plot2d(t,ysol,style=color('green'))
xlabel('x')
ylabel('y')
title('Analytical solution of ODE')
