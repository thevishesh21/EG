// Doolittle's Method (LU Decomposition)

clc;

A = input("Enter coefficient matrix A: ");
B = input("Enter RHS matrix B: ");

L = eye(3,3);
U = eye(3,3);

U(1,:) = A(1,:);

L(2,1) = A(2,1)/U(1,1);
U(2,2) = A(2,2)-(L(2,1)*U(1,2));
U(2,3) = A(2,3)-(L(2,1)*U(1,3));

L(3,1) = A(3,1)/U(1,1);
L(3,2) = (A(3,2)-(L(3,1)*U(1,2)))/U(2,2);
U(3,3) = A(3,3)-(L(3,1)*U(1,3))-(L(3,2)*U(2,3));

disp("Lower Triangular Matrix L:");
disp(L);

disp("Upper Triangular Matrix U:");
disp(U);

Y = inv(L)*B;
disp("Intermediate Matrix Y:");
disp(Y);

X = inv(U)*Y;
disp("Final Solution X:");
disp(X);
