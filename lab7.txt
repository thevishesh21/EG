// GAUSS SEIDEL METHOD

// INPUT COEFFICIENT MATRIX AND RHS MATRIX

A = input("Enter the coefficient matrix A: ");
B = input("Enter the RHS matrix B: ");

// INITIAL GUESS

X(1) = input("Enter initial value of X: ");
Y(1) = input("Enter initial value of Y: ");

tol = 0.001;

err_x = 1;
err_y = 1;

i = 1;

mprintf("\n");
mprintf("%s..%12.7f..%12.7f\n", "Initial", X(1), Y(1));

// GAUSS SEIDEL ITERATION

while (err_x > tol | err_y > tol) do

    X(i+1) = (1/A(1,1)) * (B(1) - A(1,2)*Y(i));

    Y(i+1) = (1/A(2,2)) * (B(2) - A(2,1)*X(i+1));

    mprintf("%8d..%12.7f..%12.7f\n", i, X(i+1), Y(i+1));

    err_x = abs(X(i+1) - X(i));
    err_y = abs(Y(i+1) - Y(i));

    i = i + 1;

end

// FINAL SOLUTION

disp("Final value of X:");
disp(X(i));

disp("Final value of Y:");
disp(Y(i));

// PLOTTING

j = 0:i-1;

subplot(1,2,1);
plot2d(j, X(1:i));
xtitle("X Iterations", "Iteration", "X");

subplot(1,2,2);
plot2d(j, Y(1:i));
xtitle("Y Iterations", "Iteration", "Y");
