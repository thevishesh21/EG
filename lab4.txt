// Gauss Elimination Method
clc;
clear;

// Input coefficient matrix
A = input("Enter coefficient matrix A: ");

// Input RHS matrix
B = input("Enter RHS matrix B: ");

// Augmented matrix
C = [A B];

disp("Augmented Matrix:");
disp(C);

// Forward Elimination
for i = 1:3
    for j = i+1:3

        M = C(j,i)/C(i,i);

        for k = 1:4
            C(j,k) = C(j,k) - M*C(i,k);
        end

    end
end

disp("Upper Triangular Matrix:");
disp(C);

// Back Substitution
X = zeros(3,1);

for i = 3:-1:1

    X(i) = C(i,4);

    for j = i+1:3
        X(i) = X(i) - C(i,j)*X(j);
    end

    X(i) = X(i)/C(i,i);

end

disp("Solution:");
disp(X);
